import React, { useState } from 'react';
import { View, Text, Button, StyleSheet, Image } from 'react-native';

const symbols = ['🍒', '🔔', '🍋', '⭐', '🍉'];

export default function App() {
  const [slots, setSlots] = useState(['🍒', '🍒', '🍒']);
  const [coins, setCoins] = useState(100);
  const [message, setMessage] = useState('');

  const spin = () => {
    if (coins <= 0) {
      setMessage('အသားငယ်ပြီးပါပြီ!');
      return;
    }

    const newSlots = [
      symbols[Math.floor(Math.random() * symbols.length)],
      symbols[Math.floor(Math.random() * symbols.length)],
      symbols[Math.floor(Math.random() * symbols.length)],
    ];
    setSlots(newSlots);

    if (newSlots[0] === newSlots[1] && newSlots[1] === newSlots[2]) {
      setCoins(coins + 50);
      setMessage('ဂျက်ပေါ့က်အောင်မြင်ပါပြီ! +50 coins');
    } else {
      setCoins(coins - 10);
      setMessage('ပြန်လည်ကြိုးစားပါ :) -10 coins');
    }
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>🎰 Slot Machine 🎰</Text>
      <Text style={styles.slots}>{slots.join(' | ')}</Text>
      <Text style={styles.coins}>Coins: {coins}</Text>
      <Button title="Spin" onPress={spin} />
      <Text style={styles.message}>{message}</Text>
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1, justifyContent: 'center', alignItems: 'center', backgroundColor: '#202020',
  },
  title: {
    fontSize: 32, color: '#fff', marginBottom: 20,
  },
  slots: {
    fontSize: 40, color: '#f1c40f', marginBottom: 20,
  },
  coins: {
    fontSize: 20, color: '#ecf0f1', marginBottom: 10,
  },
  message: {
    fontSize: 16, color: '#1abc9c', marginTop: 10,
  },
});
